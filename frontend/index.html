Python调用示例
import requests

resp = requests.post(
    "http://localhost:8080/api/v1/score",
    json={"features": {
        "checking_status": "no checking",
        "duration": 12,
        "credit_amount": 2000,
        "age": 35,
        ...
    }}
)
result = resp.json()
print(result["score"], result["risk_level"])
